APIs المستخدمة (مجانية — بدون مفتاح)

  1. Geocoding API — تحويل اسم المدينة إلى إحداثيات:
    https://geocoding-api.open-meteo.com/v1/search?name=CITY
  2. Forecast API — جلب الطقس الحالي بالإحداثيات:
    https://api.open-meteo.com/v1/forecast?latitude=LAT&longitude=LON¤t_weather=true
مثال دمشق: latitude=33.51&longitude=36.29&current_weather=true

متطلبات المشروع

  • حقل بحث لإدخال اسم المدينة
  • زر "بحث" لجلب الطقس
  • عرض: اسم المدينة، درجة الحرارة، سرعة الرياح، رمز الطقس
  • رسائل تحميل وخطأ واضحة للمستخدم
  • تصميم responsive يعمل على الجوال

خطوات العمل

  1. أنشئ هيكل HTML: بحث + بطاقة عرض النتائج
  2. صمّم الواجهة بـ CSS (ألوان، بطاقة، responsive)
  3. اكتب دالة searchCity(name) للـ Geocoding
  4. اكتب دالة getWeather(lat, lon) للـ Forecast
  5. اربط زر البحث بالدوال واعرض النتائج في DOM
  6. اختبر بمدن: دمشق، حلب، بيروت، عمان

JavaScript — app.js (core logic)
// DOM elements
const searchForm = document.getElementById('search-form');
const cityInput = document.getElementById('city-input');
const statusEl = document.getElementById('status');
const weatherCard = document.getElementById('weather-card');

// Search city name and return coordinates
async function searchCity(cityName) {
  const url =
    'https://geocoding-api.open-meteo.com/v1/search?name=' +
    encodeURIComponent(cityName) +
    '&count=1&language=ar&format=json';

  const response = await fetch(url);
  const data = await response.json();

  if (!data.results || data.results.length === 0) {
    throw new Error('City not found');
  }

  return data.results[0];
}

// Fetch current weather by latitude and longitude
async function getWeather(lat, lon) {
  const url =
    'https://api.open-meteo.com/v1/forecast?latitude=' +
    lat +
    '&longitude=' +
    lon +
    '¤t_weather=true';

  const response = await fetch(url);
  const data = await response.json();
  return data.current_weather;
}

// Handle form submit
searchForm.addEventListener('submit', async function (e) {
  e.preventDefault();
  const city = cityInput.value.trim();

  statusEl.textContent = 'جاري البحث...';
  statusEl.classList.remove('hidden');
  weatherCard.classList.add('hidden');

  try {
    const location = await searchCity(city);
    const weather = await getWeather(location.latitude, location.longitude);

    document.getElementById('city-name').textContent = location.name;
    document.getElementById('country-name').textContent = location.country || '';
    document.getElementById('temperature').textContent = weather.temperature + '°C';
    document.getElementById('wind-speed').textContent =
      'سرعة الرياح: ' + weather.windspeed + ' km/h';

    statusEl.classList.add('hidden');
    weatherCard.classList.remove('hidden');
  } catch (err) {
    statusEl.textContent = 'لم يتم العثور على المدينة. حاول مرة أخرى.';
  }
});
